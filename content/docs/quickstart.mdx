This guide builds one `todos.create` route from a shared contract.

The contract declares the HTTP method, path, request body, and response body once.
The server handler and client call are inferred from that declaration.

## Define A Contract

Create a shared contract file. Put it somewhere both your server and client can
import.

<Tabs>
<Tab title="Zod">

```ts contract.ts
import { router } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = router({
	todos: {
		create: {
			method: "POST",
			path: "/todos",
			body: {
				title: z.string().min(1),
			},
			response: todoSchema,
		},
	},
});
```

## Server implementation

Implement the contract on the server with the framework of your choice.

<Tabs param="server">
<Tab title="Express">

```ts twoslash
// ---cut-start---
// @filename: contract.ts
import { router } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = router({
	todos: {
		create: {
			method: "POST",
			path: "/todos",
			body: {
				title: z.string().min(1),
			},
			response: todoSchema,
		},
	},
});

// @filename: server.ts
// ---cut-end---
import { registerRoutes, router } from "@rest-rpc/express";
import express from "express";
import { api } from "./contract";

const app = express();
app.use(express.json());

const todos = new Map<string, {
	id: string;
	title: string;
	completed: boolean;
}>();

const routes = router(api, {
	todos: {
		create({ title }) {
			const todo = {
				id: crypto.randomUUID(),
				title,
				completed: false,
			};

			todos.set(todo.id, todo);

			return todo;
		},
	},
});

registerRoutes(app, routes);

app.listen(3000);
```

## Client implementation

Use the same contract to create a typed client.

<Tabs>
<Tab title="Fetch Client">

```ts twoslash
// ---cut-start---
// @filename: contract.ts
import { router } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = router({
	todos: {
		create: {
			method: "POST",
			path: "/todos",
			body: {
				title: z.string().min(1),
			},
			response: todoSchema,
		},
	},
});

// @filename: client.ts
// ---cut-end---
import { initClient } from "@rest-rpc/core";
import { api } from "./contract";

const client = initClient(api, {
	origin: "http://localhost:3000",
});

const todo = await client.todos.create.fetch({
	title: "Ship v1",
});
```

</Tab>

// ---cut-start---
// @filename: contract.ts
import { router } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = router({
	todos: {
		create: {
			method: "POST",
			path: "/todos",
			body: {
				title: z.string().min(1),
			},
			response: todoSchema,
		},
	},
});

// @filename: client.tsx
// ---cut-end---
import { initTanstackQuery } from "@rest-rpc/tanstack-query";
import { useMutation } from "@tanstack/react-query";
import { api } from "./contract";

const tq = initTanstackQuery(api, {
	origin: "http://localhost:3000",
});

const createTodo = useMutation(
	tq.todos.create.mutationOptions({
		onSuccess(response) {
			console.log(response.body.id);
		},
	}),
);

createTodo.mutate({
	title: "Ship v1",
});
