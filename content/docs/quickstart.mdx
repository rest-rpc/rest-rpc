title: Quickstart
description: Define one contract, implement it on the server, and call it from a client.
---

This guide builds one `todos.create` route from a shared contract.

<Steps>
	<Step title="Install packages">
		Add the core package, a schema library, and the adapter for the server
		framework you want to use.
	</Step>
	<Step title="Define the contract">
		Declare the HTTP method, path, request body, and response body once in a
		file that both server and client code can import.
	</Step>
	<Step title="Implement the server route">
		Use the same route tree to write a typed handler for Express, Fastify, Hono,
		NestJS, Node.js, or a fetch-native runtime.
	</Step>
	<Step title="Call it from the client">
		Create a typed fetch client or TanStack Query helpers from the same
		contract.
	</Step>
</Steps>

## Install Packages

This guide uses Zod and Express by default. The same contract works with the
other server tabs below. The validation library can be replaced with any library
implementing the [Standard Schema](https://standard-schema.dev/) or a built-in type-only schema.

<Tabs>
<Tab title="pnpm">

```bash
pnpm add @rest-rpc/core @rest-rpc/express express zod
```

</Tab>
<Tab title="npm">

```bash
npm install @rest-rpc/core @rest-rpc/express express zod
```

</Tab>
<Tab title="yarn">

```bash
yarn add @rest-rpc/core @rest-rpc/express express zod
```

</Tab>
<Tab title="bun">

```bash
bun add @rest-rpc/core @rest-rpc/express express zod
```

</Tab>
</Tabs>

If you would like to use a different backend framework, replace `@rest-rpc/express express` with the
adapter and framework package you need:

- Fastify: `@rest-rpc/fastify fastify`
- Hono: `@rest-rpc/hono hono`
- NestJS: `@rest-rpc/nest @nestjs/common @nestjs/core @nestjs/platform-express reflect-metadata rxjs`
- Node.js HTTP: `@rest-rpc/node`
- Fetch runtime `Request`/`Response`: `@rest-rpc/fetch`

For the TanStack Query client example, also install:

```bash
pnpm add @rest-rpc/tanstack-query @tanstack/react-query
```

## Define A Contract

Create a shared contract file. Put it somewhere both your server and client can
import.

<Tabs>
<Tab title="Zod">

```ts contract.ts
import { route } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};
```

</Tab>

## Implement The Contract

Implement the contract on the server with the framework of your choice.

<Tabs param="server">
<Tab title="Express">

```ts twoslash
// ---cut-start---
// @filename: contract.ts
import { route } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};

// @filename: server.ts
// ---cut-end---
import { registerRoutes, router } from "@rest-rpc/express";
import express from "express";
import { api } from "./contract";

const app = express();
app.use(express.json());

const todos = new Map<
	string,
	{
		id: string;
		title: string;
		completed: boolean;
	}
>();

const routes = router(api, {
	todos: {
		create({ body: { title } }) {
			const todo = {
				id: crypto.randomUUID(),
				title,
				completed: false,
			};

			todos.set(todo.id, todo);

			return todo;
		},
	},
});

registerRoutes(app, routes);

app.listen(3000);
```

<Tab title="NestJS">

```ts twoslash
// @experimentalDecorators: true
// ---cut-start---
// @filename: contract.ts
import { route } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};

// @filename: server.ts
// ---cut-end---
import "reflect-metadata";
import { Controller, Injectable, Module } from "@nestjs/common";
import { NestFactory } from "@nestjs/core";
import { Implement, implement, RestRpcModule } from "@rest-rpc/nest";
import { api } from "./contract";

@Injectable()
class TodoService {
	private readonly todos = new Map<
		string,
		{
			id: string;
			title: string;
			completed: boolean;
		}
	>();

	readonly routes = {
		create: implement(api).todos.create.handler(({ body: { title } }) => {
			const todo = {
				id: crypto.randomUUID(),
				title,
				completed: false,
			};

			this.todos.set(todo.id, todo);
			return todo;
		}),
	};
}

@Controller()
class TodoController {
	constructor(private readonly todos: TodoService) {}

	@Implement(api.todos)
	todosRouter() {
		return this.todos.routes;
	}
}

@Module({
	imports: [RestRpcModule.forRoot()],
	controllers: [TodoController],
	providers: [TodoService],
})
class AppModule {}

const app = await NestFactory.create(AppModule);
await app.listen(3000);
```

<Tab title="Node.js HTTP">

```ts
import { createServer } from "node:http";
import { createRouteHandler, implement } from "@rest-rpc/node";
import { api } from "./contract";

const todos = new Map();
const routes = {
	todos: {
		create: implement(api).todos.create.handler(({ body: { title } }) => {
			const todo = {
				id: crypto.randomUUID(),
				title,
				completed: false,
			};
			todos.set(todo.id, todo);
			return todo;
		}),
	},
};
const handle = createRouteHandler(routes);

const server = createServer(async (request, response) => {
	const { matched } = await handle(request, response);
	if (!matched) {
		response.writeHead(404).end("Not found");
	}
});

server.listen(3000);
```

import { createRouteHandler, implement } from "@rest-rpc/fetch";
import { api } from "./contract";

const todos = new Map();
const routes = {
	todos: {
		create: implement(api).todos.create.handler(({ body: { title } }) => {
			const todo = {
				id: crypto.randomUUID(),
				title,
				completed: false,
			};
			todos.set(todo.id, todo);
			return todo;
		}),
	},
};
const handle = createRouteHandler(routes);

export default {
	async fetch(request: Request) {
		const result = await handle(request);
		return result.matched
			? result.response
			: new Response("Not found", { status: 404 });
	},
};

## Create a Client

Use the same contract to create a typed client.

<Tabs>
<Tab title="Fetch Client">

```ts twoslash
// ---cut-start---
// @filename: contract.ts
import { route } from "@rest-rpc/core";
import { z } from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
	completed: z.boolean(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};

// @filename: client.ts
// ---cut-end---
import { initClient } from "@rest-rpc/core";
import { api } from "./contract";

const client = initClient(api, {
	baseUrl: "http://localhost:3000",
});

const response = await client.todos.create({
	body: { title: "Ship v1" },
});

if (response.status === 201) {
	const todo = response.body;
}
```

</Tab>
