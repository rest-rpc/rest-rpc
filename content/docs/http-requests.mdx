## Request With JSON Body

```ts
export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
					labels: z.array(z.string()).optional(),
				}),
			)
			.response(201, todoSchema),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		create({ body: { title, labels } }) {
			return createTodo({ title, labels });
		},
	},
});
```

```ts
const response = await client.todos.create({
	body: { title: "Write docs", labels: ["docs"] },
});

if (response.status === 201) {
	const todo = response.body;
}
```

## Request Without Body

Declare a request without a body by omitting a body setter.

```ts
export const api = {
	todos: {
		list: route
			.get("/todos")
			.query(
				z.object({
					limit: z.number().optional(),
				}),
			)
			.response(200, z.array(todoSchema)),
		byId: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, todoSchema),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		list({ query: { limit } }) {
			return listTodos({ limit });
		},
		byId({ params: { id } }) {
			return getTodoById({ id });
		},
	},
});
```

## Request With Form Body

Use `.formBody()` for `application/x-www-form-urlencoded` request bodies.
The body is sent as `URLSearchParams` and then passed to the handler as a
validated object under `body`.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	forms: {
		submit: route
			.post("/forms")
			.formBody(
				z.object({
					title: z.string(),
					count: z.coerce.number<number>(),
					remember: z.string().optional(),
					tags: z.array(z.string()).optional(),
				}),
			)
			.response(
				201,
				z.object({
					title: z.string(),
					count: z.number(),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	forms: {
		submit({ body }) {
			return {
				title: body.title,
				count: body.count,
			};
		},
	},
});
```

```ts
const response = await client.forms.submit({
	body: {
		title: "Write docs",
		count: 3,
		tags: ["typescript", "rpc"],
	},
});

if (response.status === 201) {
	const result = response.body;
}
```

Body parsers should pass a `URLSearchParams` instance to rest-rpc for form bodies.

## Request With Multipart Body

Use `.multipartBody()` for `multipart/form-data` request bodies.
The body is sent as `FormData` and then passed to the handler as a validated
object under `body`.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	uploads: {
		create: route
			.post("/uploads")
			.multipartBody(
				z.object({
					title: z.string(),
					file: z.instanceof(File),
					tags: z.array(z.string()).optional(),
				}),
			)
			.response(
				201,
				z.object({
					title: z.string(),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	uploads: {
		create({ body }) {
			return {
				title: body.title,
			};
		},
	},
});
```

```ts
const response = await client.uploads.create({
	body: {
		title: "Write docs",
		file,
		tags: ["typescript", "rpc"],
	},
});

if (response.status === 201) {
	const result = response.body;
}
```

Body parsers should pass a `FormData` instance to rest-rpc for multipart bodies.

## Request With Custom Content Type

Custom request bodies are declared with `.customBody()`. The complete payload
is exposed under `body` at the client call and server handler.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	imports: {
		csv: route
			.post("/imports.csv")
			.customBody({
				contentType: "text/csv",
				schema: z.string(),
			})
			.response(
				201,
				z.object({
					imported: z.number(),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	imports: {
		csv({ body }) {
			return importCsv(body);
		},
	},
});
```

```ts
const response = await client.imports.csv({
	body: "id,title\n1,Write docs\n",
});

if (response.status === 201) {
	const result = response.body;
}
```

## Request With Multiple Custom Content Types

Pass an array of content types to `.customBody()` to declare multiple custom
request content types.
The `body` value contains both the selected `contentType` and the `payload`.

```ts
export const api = {
	images: {
		upload: route
			.post("/images")
			.customBody({
				contentType: ["image/png", "image/jpeg"],
				schema: z.instanceof(Uint8Array),
			})
			.response(
				201,
				z.object({
					id: z.string(),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	images: {
		upload({ body }) {
			return saveImage({
				contentType: body.contentType,
				bytes: body.payload,
			});
		},
	},
});
```

```ts
const response = await client.images.upload({
	body: {
		contentType: "image/png",
		payload: bytes,
	},
});

if (response.status === 201) {
	const image = response.body;
}
```

## Request With Fetch-Managed Content Type

Omit `contentType` when the request body should be passed directly to `fetch`
and the runtime should decide the `content-type` header.

```ts
export const api = {
	forms: {
		submit: route
			.post("/forms")
			.customBody(z.instanceof(URLSearchParams))
			.response(
				201,
				z.object({
					title: z.string(),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	forms: {
		submit({ body }) {
			return {
				title: body.get("title") ?? "",
			};
		},
	},
});
```

```ts
const response = await client.forms.submit({
	body: new URLSearchParams({
		title: "Write docs",
	}),
});

if (response.status === 201) {
	const result = response.body;
}
```
