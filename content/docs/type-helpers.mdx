---
title: Type Helpers
description: Infer route types for clients, TanStack Query, and server handlers
---

Type helpers take a route declaration and infer the type used at a specific
runtime boundary.

```ts
import { route } from "@rest-rpc/core";
import { z } from "zod";

export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.query(z.object({ includeDone: z.boolean().optional() }))
			.response(200, z.object({ id: z.string(), title: z.string() }))
			.response(404, z.object({ code: z.literal("NOT_FOUND") })),
		page: route
			.get("/todos/page")
			.query(
				z.object({
					cursor: z.string().optional(),
					status: z.enum(["open", "done"]),
					limit: z.number(),
				}),
			)
			.response(
				200,
				z.object({
					items: z.array(z.object({ id: z.string(), title: z.string() })),
					nextCursor: z.string().optional(),
				}),
			),
		create: route
			.post("/todos")
			.body(z.object({ title: z.string() }))
			.response(201, z.object({ id: z.string(), title: z.string() })),
		events: route
			.get("/todos/events")
			.streamResponse(200, z.object({ id: z.string(), message: z.string() })),
	},
};
```

## Fetch Client

Import fetch client helpers from `@rest-rpc/core`.

```ts
import type { ClientRequest, ClientResponse } from "@rest-rpc/core";

type GetTodoRequest = ClientRequest<typeof api.todos.get>;
// { params: { id: string }; query: { includeDone?: boolean } }

type GetTodoResponse = ClientResponse<typeof api.todos.get>;
// | { status: 200; body: { id: string; title: string }; headers: Headers }
// | { status: 404; body: { code: "NOT_FOUND" }; headers: Headers }
```

`ClientResponse` is the result returned when a client route is called.

## Server

Import server helpers from the server adapter you use. The adapter-specific
`RouteRequest` includes that adapter's handler context.

```ts
import type {
	RouteErrors,
	RouteRequest,
	RouteRequestData,
	RouteResponse,
} from "@rest-rpc/express";

type GetTodoRequest = RouteRequest<typeof api.todos.get>;
// { params: { id: string }; query: { includeDone?: boolean }; context: HttpRouteHandlerContext }

type GetTodoRequestData = RouteRequestData<typeof api.todos.get>;
// { params: { id: string }; query: { includeDone?: boolean } }

type GetTodoResponse = RouteResponse<typeof api.todos.get>;
// | { status: 200; body: { id: string; title: string } }
// | { status: 404; body: { code: "NOT_FOUND" } }

type GetTodoErrors = RouteErrors<typeof api.todos.get>;
// { status: 404; body: { code: "NOT_FOUND" } }
```

`RouteResponse` is the response envelope returned by a declared HTTP handler.
Responses with a body include `status` and `body`; responses declared without a
body include only `status` (plus `responseHeaders` when declared).

## TanStack Query

Import TanStack Query helpers from `@rest-rpc/tanstack-query`.

```ts
import type {
	RouteQueryData,
	RouteInfiniteQueryData,
	RouteStreamedQueryData,
	RouteQueryError,
	RouteMutationVariables,
} from "@rest-rpc/tanstack-query";

type GetTodoData = RouteQueryData<typeof api.todos.get>;
// { status: 200; body: { id: string; title: string }; headers: Headers }

type TodoPagesData = RouteInfiniteQueryData<typeof api.todos.page>;
// {
//   pages: Array<{
//     status: 200;
//     body: {
//       items: Array<{ id: string; title: string }>;
//       nextCursor?: string;
//     };
//     headers: Headers;
//   }>;
//   pageParams: Array<{
//     query: { cursor?: string; status: "open" | "done"; limit: number };
//   }>;
// }

type TodoEventsData = RouteStreamedQueryData<typeof api.todos.events>;
// Array<{ id: string; message: string }>

type GetTodoError = RouteQueryError<typeof api.todos.get>;
// | { status: 404; body: { code: "NOT_FOUND" }; headers: Headers }
// | Error

type CreateTodoVariables = RouteMutationVariables<typeof api.todos.create>;
// { body: { title: string } }
```
