## Install

```sh
pnpm add @rest-rpc/hono
```

## Usage

Register routes on a Hono application or a child app. Hono middleware and the native
context remain available to handlers.

```ts
import { Hono } from "hono";
import { route, registerRoutes } from "@rest-rpc/hono";
import { z } from "zod";

const app = new Hono();
const api = new Hono();
const todos = new Map([["todo_1", { id: "todo_1", title: "Write docs" }]]);

api.use("*", async (c, next) => {
	console.log(c.req.method, c.req.path);
	await next();
});

const getTodo = route
	.get("/todos/:id")
	.params(z.object({ id: z.string() }))
	.handler(({ params, c }) => {
		const todo = todos.get(params.id);
		return todo
			? {
					status: 200,
					body: todo,
					responseHeaders: { "x-request-id": c.req.header("x-request-id") },
				}
			: { status: 404, body: { code: "TODO_NOT_FOUND" as const } };
	});

export const routes = { todos: { get: getTodo } };

registerRoutes(api, routes);
app.route("/api", api);
export default app;
```

## Framework Context

Provide request-scoped context with the `context` option. Augment
`DefaultContext` to type it across handlers:

```ts
declare module "@rest-rpc/hono" {
	interface DefaultContext {
		user: { id: string };
	}
}
```

The context factory receives the Hono context. Handlers receive the following
adapter fields and typed context:

```ts
type RouteHandlerFields<E extends Env = Env> = {
	c: Context<E>;
	signal: AbortSignal;
	context: DefaultContext;
};
```

```ts
registerRoutes(app, routes, {
	context: ({ c }) => ({
		user: { id: c.req.header("x-user-id") ?? "" },
	}),
});
```

See [Middleware](/docs/middleware) for route middleware that shares this context.

## Options

```ts
type RegisterRoutesOptions<TEnv extends Env = Env> = {
	requestValidationErrorHandler?: RequestValidationErrorHandler<TEnv>;
	responseValidationErrorHandler?: ResponseValidationErrorHandler<TEnv>;
	bodyCodecs?: readonly BodyCodec<HonoRequest>[];
	requestBodyLimit?: number;
	middleware?: Array<
		(c: Context<TEnv>, next: Next, route: RouteDeclaration) => any
	>;
};
```

`bodyCodecs`: see [Serialization and Codecs](/docs/http-behavior/serialization).

`requestBodyLimit`: see [request body limits](/docs/http-behavior/serialization#server-request-deserialization).

### Error handling

Use `requestValidationErrorHandler` and `responseValidationErrorHandler` to
customize [validation errors](/docs/http-behavior/schemas). Other errors use Hono's error handler.

```ts
registerRoutes(app, routes, {
	requestValidationErrorHandler: (error, c) => {
		return c.json({ code: "VALIDATION_ERROR", issues: error.issues }, 422);
	},
	responseValidationErrorHandler: (_error, c) => {
		return c.json({ code: "INVALID_RESPONSE" }, 500);
	},
});

app.onError((error, c) => {
	return c.json({ code: "INTERNAL_SERVER_ERROR" }, 500);
});
```

### Middleware

```ts
registerRoutes(app, routes, {
	middleware: [
		async (c, next, route) => {
			const authorization = c.req.header("authorization");
			if (!authorization) {
				return c.json({ code: "UNAUTHORIZED" }, 401);
			}
			await next();
		},
	],
});
```
