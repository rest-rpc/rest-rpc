## Install

```sh
pnpm add @rest-rpc/hono
```

## Usage

```ts
import { implement, registerRoutes } from "@rest-rpc/hono";
import { Hono } from "hono";
import { api } from "./contract";

const app = new Hono();

const implementor = implement(api);

const routes = {
	todos: {
		list: implementor.todos.list.handler(() => listTodos()),
		get: implementor.todos.get.handler(({ params: { id }, c }) => {
			const authorization = c.req.header("authorization");
			return getTodo(id, { authorization });
		}),
		create: implementor.todos.create.handler(({ body: { title } }) => {
			return createTodo({ title });
		}),
	},
};

registerRoutes(app, routes);

export default app;
```

## Framework Context

Handlers receive the following adapter fields directly:

```ts
type RouteHandlerFields<E extends Env = Env> = {
	c: Context<E>;
	signal: AbortSignal;
};
```

## Options

```ts
type RegisterRoutesOptions<TEnv extends Env = Env> = {
	requestValidationErrorHandler?: RequestValidationErrorHandler<TEnv>;
	responseValidationErrorHandler?: ResponseValidationErrorHandler<TEnv>;
	bodyParser?: HonoBodyParser;
	middleware?: Array<
		(c: Context<TEnv>, next: Next, route: RouteDeclaration) => any
	>;
};
```

### Body Parsing

```ts
registerRoutes(app, routes, {
	bodyParser: (request) => request.json(),
});
```

### Error handling

Request validation errors default to a 400 response. Response contract
validation errors default to a generic 500 response. Customize them with
handlers that receive the native Hono context. Other errors are handled by
Hono's error handler.

```ts
registerRoutes(app, routes, {
	requestValidationErrorHandler: (error, c) => {
		return c.json({ code: "VALIDATION_ERROR", issues: error.issues }, 422);
	},
	responseValidationErrorHandler: (_error, c) => {
		return c.json({ code: "INVALID_RESPONSE" }, 500);
	},
});

app.onError((error, c) => {
	return c.json({ code: "INTERNAL_SERVER_ERROR" }, 500);
});
```

### Middleware

```ts
registerRoutes(app, routes, {
	middleware: [
		async (c, next, route) => {
			const authorization = c.req.header("authorization");
			if (!authorization) {
				return c.json({ code: "UNAUTHORIZED" }, 401);
			}
			await next();
		},
	],
});
```
