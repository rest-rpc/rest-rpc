## Install

```sh
pnpm add @rest-rpc/nest
```

## Usage

Use `@Implement()` decorator on a Nest controller method with a contract route or tree.
The method should return the matching completed route implementation or tree.

```ts
import { Controller, Injectable, Module } from "@nestjs/common";
import { NestFactory } from "@nestjs/core";
import { Implement, implement, RestRpcModule } from "@rest-rpc/nest";
import { api } from "./contract";

const builders = implement(api);

@Injectable()
class TodoRepository {
	find(id: string) {
		return { id, title: "Write docs" };
	}

	create(title: string) {
		return { id: "todo-1", title };
	}
}

@Injectable()
class TodoRoutes {
	constructor(private readonly todos: TodoRepository) {}

	readonly routes = {
		get: builders.todos.get.handler(({ params: { id } }) =>
			this.todos.find(id),
		),
		create: builders.todos.create.handler(({ body: { title } }) => ({
			status: 201,
			body: this.todos.create(title),
		})),
	};
}

@Controller()
class TodoController {
	constructor(private readonly todos: TodoRoutes) {}

	@Implement(api.todos)
	todosRoutes() {
		return this.todos.routes;
	}
}

@Module({
	imports: [RestRpcModule.forRoot()],
	controllers: [TodoController],
	providers: [TodoRepository, TodoRoutes],
})
class AppModule {}

const app = await NestFactory.create(AppModule);
await app.listen(3000);
```

The decorated method can return a matching implementation from any
source. `@Implement()` uses its argument to register Nest routes and uses the
method return value when handling a request.

Nest decorators applied to the decorated method are copied to every generated
route method. This allows guards, interceptors, and Nest parameter decorators
to be used at the contract-tree boundary.

## Single Routes

The same decorator accepts a single explicit HTTP route:

```ts
@Controller()
class TodoController {
	constructor(private readonly todos: TodoRepository) {}

	@Implement(api.todos.get)
	getTodo() {
		return implement(api.todos.get).handler(({ params: { id } }) =>
			this.todos.find(id),
		);
	}
}
```

:::warning
If you're using [Procedure routes](docs/procedures), which automatically
generate the path from the tree keys, you must use a tree with both
the decorator and the return value. Otherwise, the decorator cannot determine the path to register with Nest.
:::

## Server-First Routes

`@Implement()` also accepts server-first route declarations. Keep the
declaration outside the method so the decorator can read its method and path
when Nest registers the controller.

```ts
import { Implement, route } from "@rest-rpc/nest";
import { z } from "zod";

const todoRoutes = {
	get: route.get("/todos/:id").params(z.object({ id: z.string() })),
};

@Controller()
class TodoController {
	constructor(private readonly todos: TodoRepository) {}

	@Implement(todoRoutes)
	routes() {
		return {
			get: todoRoutes.get.handler(({ params: { id } }) => this.todos.find(id)),
		};
	}
}
```

Completed server-first implementations can be returned in the same way. The
decorator only requires a route or tree from which Nest registration metadata
can be read.

## Framework Context

Nest handlers receive `executionContext` and `signal` directly. Application
context returned by `createContext` is available under `context`.

```ts
const get = implement(api).todos.get.handler(
	({ context, executionContext, params: { id }, signal }) => {
		return findTodo(id, context.userId, executionContext, signal);
	},
);
```

Augment `DefaultContext` to type application context across Nest handlers:

```ts
import type { Request } from "express";
import { RestRpcModule } from "@rest-rpc/nest";

declare module "@rest-rpc/nest" {
	interface DefaultContext {
		request: Request;
		userId: string;
	}
}

RestRpcModule.forRoot({
	createContext: (context) => {
		const request = context.switchToHttp().getRequest<Request>();
		return {
			request,
			userId: request.header("x-user-id") ?? "",
		};
	},
});
```

## Options

```ts
type RestRpcModuleOptions<TContext = DefaultContext> = {
	createContext?: (context: ExecutionContext) => TContext | Promise<TContext>;
};
```

### Error handling

Request validation throws `RequestValidationException`, which defaults to a
400 response. Response contract validation throws
`ResponseValidationException`, which defaults to a generic 500 response. Both
are Nest `HttpException` subclasses and can be customized with ordinary Nest
exception filters. Other errors continue through Nest's exception handling
unchanged.

```ts
import { ArgumentsHost, Catch, ExceptionFilter } from "@nestjs/common";
import { APP_FILTER, HttpAdapterHost } from "@nestjs/core";
import { RequestValidationException } from "@rest-rpc/nest";

@Catch(RequestValidationException)
class RequestValidationFilter implements ExceptionFilter {
	constructor(private readonly adapterHost: HttpAdapterHost) {}

	catch(error: RequestValidationException, host: ArgumentsHost) {
		const response = host.switchToHttp().getResponse();
		this.adapterHost.httpAdapter.reply(
			response,
			{
				code: "VALIDATION_ERROR",
				issues: error.validationError.issues,
			},
			422,
		);
	}
}

@Module({
	imports: [RestRpcModule.forRoot()],
	providers: [{ provide: APP_FILTER, useClass: RequestValidationFilter }],
})
class AppModule {}
```
