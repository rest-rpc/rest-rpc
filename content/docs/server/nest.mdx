## Install

```sh
pnpm add @rest-rpc/nest
```

## Usage

The Nest adapter lets a contract router map to a normal Nest provider class.
The controller declares the Nest route boundary, while the provider implements
the contract.

```ts
import { Controller, Injectable, Module } from "@nestjs/common";
import { NestFactory } from "@nestjs/core";
import {
	initNest,
	RestRpcModule,
	RouteRequest,
	RouteHandlers,
	Router,
} from "@rest-rpc/nest";
import type { Request, Response } from "express";
import { api } from "./contract";

type AppContext = {
	userId: string;
};

const nest = initNest<AppContext>();

@Injectable()
class TodoRepository {
	findForUser(id: string, userId: string) {
		return { id, title: "Write docs", userId };
	}

	createForUser(input: { title: string; userId: string }) {
		return { id: "todo-1", title: input.title, userId: input.userId };
	}
}

@Injectable()
class TodoService implements RouteHandlers<typeof api.todos, AppContext> {
	constructor(private readonly todos: TodoRepository) {}

	get({ id, context }: RouteRequest<typeof api.todos.get, AppContext>) {
		return this.todos.findForUser(id, context.userId);
	}

	create({ title, context }: RouteRequest<typeof api.todos.create, AppContext>) {
		return {
			status: 201 as const,
			body: this.todos.createForUser({ title, userId: context.userId }),
		};
	}
}

@Controller()
class TodoController {
	constructor(private readonly todos: TodoService) {}

	@Router(api.todos)
	todosRouter() {
		return nest.router(api.todos, this.todos);
	}
}

@Module({
	imports: [
		RestRpcModule.forRoot<AppContext, Request, Response>({
			createContext: ({ req }) => ({
				userId: String(req.headers["x-user-id"] ?? ""),
			}),
		}),
	],
	controllers: [TodoController],
	providers: [TodoRepository, TodoService],
})
class AppModule {}

const app = await NestFactory.create(AppModule);
await app.listen(3000);
```

`@Router()` belongs on the controller because it tells Nest which HTTP routes to
register. `router()` connects the contract router to the provider instance that
implements it.

The provider is a first-class Nest provider. It can use constructor injection,
module ownership, scopes, lifecycle hooks, and testing overrides like any other
service.

## Single Routes

Use `@Route()` when one endpoint needs custom glue, or when existing providers
do not match the contract shape.

```ts
import { Controller } from "@nestjs/common";
import { Route, initNest } from "@rest-rpc/nest";
import { api } from "./contract";

type AppContext = {
	userId: string;
};

const nest = initNest<AppContext>();

@Controller()
class TodoController {
	constructor(private readonly todos: TodoRepository) {}

	@Route(api.todos.get)
	getTodo() {
		return nest.route(api.todos.get, ({ id, context }) => {
			return this.todos.findForUser(id, context.userId);
		});
	}
}
```

## Inline Routers

`@Router()` can also return an inline handler object. This is useful for small
examples, tests, and prototypes, but provider-backed routers are usually a
better fit for production Nest modules.

```ts
@Controller()
class TodoController {
	@Router(api.todos)
	todosRouter() {
		return nest.router(api.todos, {
			get: ({ id }) => ({ id, title: "Write docs" }),
			create: ({ title }) => ({
				status: 201 as const,
				body: { id: "todo-1", title },
			}),
		});
	}
}
```

## Framework Context

Nest route handlers receive the context returned by `createContext`, plus the
raw request, response, and abort signal.

```ts
type NestHandlerContext<TContext, TRequest = unknown, TResponse = unknown> =
	TContext & {
		req: TRequest;
		res: TResponse;
		signal: AbortSignal;
	};
```

You can pass request and response types to `initNest` or the module options when
you want framework-specific access.

```ts
import type { Request, Response } from "express";
import { initNest, RestRpcModule } from "@rest-rpc/nest";

type AppContext = {
	userId: string;
};

const nest = initNest<AppContext, Request, Response>();

RestRpcModule.forRoot<AppContext, Request, Response>({
	createContext: ({ req }) => ({
		userId: req.header("x-user-id") ?? "",
	}),
});
```

## Options

```ts
type RestRpcModuleOptions<TContext, TRequest = unknown, TResponse = unknown> = {
	createContext?: (input: {
		req: TRequest;
		res: TResponse;
		signal: AbortSignal;
		executionContext: ExecutionContext;
	}) => TContext | Promise<TContext>;
	errorHandlers?: ServerErrorHandlers<TContext>;
};
```

### Error Handlers

Request validation errors use `onRequestValidationError`. Response contract
validation errors use `onResponseValidationError` and default to a generic 500
response. Other unhandled route errors use `onUnhandledError`, or are re-thrown
when that hook is omitted or returns `undefined`.

```ts
RestRpcModule.forRoot<AppContext>({
	errorHandlers: {
		onRequestValidationError: ({ issues }) => ({
			status: 422,
			body: { code: "VALIDATION_ERROR", issues },
		}),
		onResponseValidationError: () => ({
			status: 500,
			body: { code: "INVALID_RESPONSE" },
		}),
		onUnhandledError: () => ({
			status: 500,
			body: { code: "INTERNAL_SERVER_ERROR" },
		}),
	},
});
```

:::note
The current Nest adapter is an HTTP adapter. It reuses Nest controller routing
and a global interceptor to execute rest-rpc route implementations.
:::
