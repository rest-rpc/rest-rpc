## Install

```package-install
npm i @rest-rpc/nest
```

## Usage

```ts
import "reflect-metadata";
import { Controller, Injectable, Module } from "@nestjs/common";
import { NestFactory } from "@nestjs/core";
import { Implement, route, RestRpcModule } from "@rest-rpc/nest";
import { z } from "zod";

const getDeclaration = route
	.get("/todos/:id")
	.params(z.object({ id: z.string() }));
const createDeclaration = route.input(z.object({ title: z.string() }));

const declarations = {
	todos: { get: getDeclaration, create: createDeclaration },
};

@Injectable()
class TodoRepository {
	private readonly todos = new Map<string, { id: string; title: string }>();

	find(id: string) {
		return this.todos.get(id);
	}

	create(title: string) {
		const todo = { id: crypto.randomUUID(), title };
		this.todos.set(todo.id, todo);
		return todo;
	}
}

@Injectable()
class TodoRoutes {
	constructor(private readonly todos: TodoRepository) {}

	readonly get = getDeclaration.handler(({ params }) => {
		const todo = this.todos.find(params.id);
		return todo
			? { status: 200, body: todo }
			: { status: 404, body: { code: "TODO_NOT_FOUND" as const } };
	});

	readonly create = createDeclaration.handler(({ input: { title } }) =>
		this.todos.create(title),
	);

	readonly routes = { todos: { get: this.get, create: this.create } };
}

@Controller()
class TodoController {
	constructor(private readonly todos: TodoRoutes) {}

	@Implement(declarations)
	todoRoutes() {
		return this.todos.routes;
	}
}

@Module({
	imports: [RestRpcModule.forRoot()],
	controllers: [TodoController],
	providers: [TodoRepository, TodoRoutes],
})
class AppModule {}

const app = await NestFactory.create(AppModule);
await app.listen(3000);
```

`@Implement()` registers the methods and paths from its argument. The decorated
method returns the matching completed route tree.

Nest decorators on the decorated method are copied to every generated route
method, so guards, interceptors, and parameter decorators can be applied to a
whole route tree.

### Single Route Registration

The decorator also accepts a single route with an explicit path:

```ts
const getTodo = route.get("/todos/:id").params(z.object({ id: z.string() }));

@Controller()
class TodoController {
	constructor(private readonly todos: TodoRepository) {}

	@Implement(getTodo)
	get() {
		return getTodo.handler(({ params }) => this.todos.find(params.id));
	}
}
```

:::warning
Route with [derived path](/docs/route-builder#start-with-defaults) is not supported.
Add explicit method and path or use a route tree with `@Implement()` instead.
:::

## Framework Context

Handlers and middleware receive the Nest `executionContext` in addition to the
[common handler arguments](/docs/route-builder#handler-arguments). Use it to
read the underlying request or response:

```ts
readonly get = getDeclaration.handler(({ params, executionContext }) => {
	const request = executionContext.switchToHttp().getRequest();
	console.log(request.headers["x-request-id"]);
	return this.todos.find(params.id);
});
```

To share an application [context](/docs/context) type across routes, augment
`DefaultContext` in `@rest-rpc/nest`.

## Options

Pass options to `RestRpcModule.forRoot()`:

```ts
type RestRpcModuleOptions = {
	bodyCodecs?: readonly BodyCodec<unknown>[];
	disableRequestValidation?: boolean;
	disableResponseValidation?: boolean;
};
```

- `bodyCodecs`: see [Request and Response Bodies](#request-and-response-bodies).
- `disableRequestValidation`, `disableResponseValidation`: see
  [Disabling Server Validation](/docs/http-behavior/schemas#disabling-server-validation).

## Request and Response Bodies

Request bodies are parsed by the underlying Express or Fastify platform.
Configure its body parsers for the content types your routes accept; rest-rpc
validates the resulting request `body`. A custom codec's `deserialize` receives
the platform's request object, typed as `unknown`, and replaces the parsed body
for its content types.

JSON responses go through Nest's normal serialization. Text, form, and binary
responses accept the same values as the [Express adapter](/docs/server/express#request-and-response-bodies)
and are sent as a `StreamableFile`. A custom serializer's `body` is returned to
Nest as the handler result, so return a `StreamableFile` for binary or streamed
output.

## Error Handling

Validation failures are thrown as `RequestValidationException` (400) and
`ResponseValidationException` (500). Both are Nest `HttpException` subclasses
whose response is the [default validation error response](/docs/http-behavior/errors),
and both expose the original error as `validationError`. Replace the response
with ordinary Nest exception filters. Other errors also use Nest's exception
handling.

```ts
import { ArgumentsHost, Catch, ExceptionFilter } from "@nestjs/common";
import { APP_FILTER, HttpAdapterHost } from "@nestjs/core";
import { RequestValidationException } from "@rest-rpc/nest";

@Catch(RequestValidationException)
class RequestValidationFilter implements ExceptionFilter {
	constructor(private readonly adapterHost: HttpAdapterHost) {}

	catch(error: RequestValidationException, host: ArgumentsHost) {
		const response = host.switchToHttp().getResponse();
		this.adapterHost.httpAdapter.reply(
			response,
			{
				code: "VALIDATION_ERROR",
				issues: error.validationError.issues,
			},
			422,
		);
	}
}

@Module({
	imports: [RestRpcModule.forRoot()],
	providers: [{ provide: APP_FILTER, useClass: RequestValidationFilter }],
})
class AppModule {}
```
