## Install

```sh
pnpm add @rest-rpc/fetch
```

## Usage

Create a matcher and call it from your runtime's Fetch
entry point. The result contains a `Response` when a route matched.
You choose how to handle unmatched requests.

```ts
import { route, createRouteHandler } from "@rest-rpc/fetch";
import { z } from "zod";

const todos = new Map([["todo_1", { id: "todo_1", title: "Write docs" }]]);

const getTodo = route
	.get("/todos/:id")
	.params(z.object({ id: z.string() }))
	.handler(({ params, request }) => {
		console.log(request.headers.get("x-request-id"));
		const todo = todos.get(params.id);
		return todo
			? { status: 200, body: todo }
			: { status: 404, body: { code: "TODO_NOT_FOUND" as const } };
	});

export const routes = { todos: { get: getTodo } };

const handler = createRouteHandler(routes);

export default {
	async fetch(request: Request) {
		const result = await handler(request);
		return result.matched
			? result.response
			: new Response("Not found", { status: 404 });
	},
};
```

## Framework Context

Fetch handlers can receive user-provided application context. Augment
`DefaultContext` to type it across handlers:

```ts
declare module "@rest-rpc/fetch" {
	interface DefaultContext {
		user: { id: string };
	}
}
```

Handlers receive the following adapter fields and typed context:

```ts
type RouteHandlerFields = {
	request: Request;
	signal: AbortSignal;
	context: DefaultContext;
};
```

Pass the context as the second argument to `handler`:

```ts
const result = await handler(request, { user });
```

## Options

```ts
type CreateFetchHandlerOptions = {
	bodyCodecs?: readonly BodyCodec<Request>[];
	requestBodyLimit?: number;
	requestValidationErrorHandler?: RequestValidationErrorHandler;
	responseValidationErrorHandler?: ResponseValidationErrorHandler;
};
```

`bodyCodecs`: see [Serialization and Codecs](/docs/http-behavior/serialization).

`requestBodyLimit`: see [request body limits](/docs/http-behavior/serialization#server-request-deserialization).

### Error handling

Use `requestValidationErrorHandler` and `responseValidationErrorHandler` to
customize [validation errors](/docs/http-behavior/schemas). Other errors reject the handler promise.

```ts
const handler = createRouteHandler(routes, {
	requestValidationErrorHandler: (error, request) => {
		return Response.json(
			{ code: "VALIDATION_ERROR", issues: error.issues },
			{ status: 422 },
		);
	},
	responseValidationErrorHandler: (_error, request) => {
		return Response.json({ code: "INVALID_RESPONSE" }, { status: 500 });
	},
});

export default {
	async fetch(request: Request) {
		try {
			const result = await handler(request);
			return result.matched
				? result.response
				: new Response("Not found", { status: 404 });
		} catch (error) {
			return Response.json({ code: "INTERNAL_SERVER_ERROR" }, { status: 500 });
		}
	},
};
```
