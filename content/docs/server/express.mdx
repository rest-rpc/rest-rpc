## Install

```sh
pnpm add @rest-rpc/express
```

## Usage

```ts
import { registerRoutes, router } from "@rest-rpc/express";
import express from "express";
import { api } from "./contract";

const app = express();

app.use(express.json());

const routes = router(api, {
	todos: {
		list() {
			return listTodos();
		},
		get({ params: { id }, context }) {
			const authorization = context.req.header("authorization");
			return getTodo(id, { authorization });
		},
		create({ body: { title } }) {
			return createTodo({ title });
		},
	},
});

registerRoutes(app, routes);

app.listen(3000);
```

type HttpRouteHandlerContext = {

## Options

```ts
type RegisterRoutesOptions = {
	requestValidationErrorHandler?: RequestValidationErrorHandler;
	responseValidationErrorHandler?: ResponseValidationErrorHandler;
	middleware?: Array<
		(
			req: Request,
			res: Response,
			next: NextFunction,
			route: HttpRouteDeclaration,
		) => unknown
	>;
};
```

### Error handling

Request validation errors default to a 400 response. Response contract
validation errors default to a generic 500 response. Customize them with
handlers that receive native Express arguments. Other errors are passed to the
next Express error handler.

```ts
registerRoutes(app, routes, {
	requestValidationErrorHandler: (error, req, res) => {
		res.status(422).json({ code: "VALIDATION_ERROR", issues: error.issues });
	},
	responseValidationErrorHandler: (_error, req, res) => {
		res.status(500).json({ code: "INVALID_RESPONSE" });
	},
});

app.use((error, req, res, next) => {
	res.status(500).json({ code: "INTERNAL_SERVER_ERROR" });
});
```

### Middleware

```ts
registerRoutes(app, routes, {
	middleware: [
		(req, res, next, route) => {
			const authorization = req.header("authorization");
			if (!authorization) {
				res.status(401).send({ code: "UNAUTHORIZED" });
				return;
			}
			next();
		},
	],
});
```
