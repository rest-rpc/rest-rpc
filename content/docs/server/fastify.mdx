## Install

```sh
pnpm add @rest-rpc/fastify
```

## Usage

Register routes inside a Fastify plugin to keep dependencies and hooks scoped to
a module. Handlers can access the native request and reply.

```ts
import Fastify from "fastify";
import { route, registerRoutes } from "@rest-rpc/fastify";
import { z } from "zod";

const app = Fastify({ logger: true });
const todos = new Map([["todo_1", { id: "todo_1", title: "Write docs" }]]);

const getTodo = route
	.get("/todos/:id")
	.params(z.object({ id: z.string() }))
	.handler(({ params, req }) => {
		req.log.info({ todoId: params.id }, "Reading todo");
		const todo = todos.get(params.id);
		return todo
			? { status: 200, body: todo }
			: { status: 404, body: { code: "TODO_NOT_FOUND" as const } };
	});

export const routes = { todos: { get: getTodo } };

await app.register(
	async (api) => {
		registerRoutes(api, routes);
	},
	{ prefix: "/api" },
);

await app.listen({ port: 3000 });
```

## Framework Context

Provide request-scoped context with the `context` option. Augment
`DefaultContext` to type it across handlers:

```ts
declare module "@rest-rpc/fastify" {
	interface DefaultContext {
		user: { id: string };
	}
}
```

The context factory receives the Fastify request and reply. Handlers receive
the following adapter fields and typed context:

```ts
type RouteHandlerFields = {
	req: FastifyRequest;
	reply: FastifyReply;
	signal: AbortSignal;
	context: DefaultContext;
};
```

```ts
registerRoutes(app, routes, {
	context: ({ req }) => ({
		user: { id: req.headers["x-user-id"]?.toString() ?? "" },
	}),
});
```

See [Middleware](/docs/middleware) for route middleware that shares this context.

## Options

```ts
type RegisterRoutesOptions = {
	bodyCodecs?: readonly BodyCodec<FastifyRequest>[];
	requestValidationErrorHandler?: RequestValidationErrorHandler;
	responseValidationErrorHandler?: ResponseValidationErrorHandler;
	preHandler?: Array<
		(
			req: FastifyRequest,
			reply: FastifyReply,
			route: RouteDeclaration,
		) => unknown
	>;
};
```

`bodyCodecs`: see [Serialization and Codecs](/docs/http-behavior/serialization).

### Error handling

Use `requestValidationErrorHandler` and `responseValidationErrorHandler` to
customize [validation errors](/docs/http-behavior/schemas). Other errors use Fastify's error handler.

```ts
registerRoutes(app, routes, {
	requestValidationErrorHandler: (error, request, reply) => {
		return reply.status(422).send({
			code: "VALIDATION_ERROR",
			issues: error.issues,
		});
	},
	responseValidationErrorHandler: (_error, request, reply) => {
		return reply.status(500).send({ code: "INVALID_RESPONSE" });
	},
});

app.setErrorHandler((error, request, reply) => {
	return reply.status(500).send({ code: "INTERNAL_SERVER_ERROR" });
});
```

### preHandler

```ts
registerRoutes(app, routes, {
	preHandler: [
		async (req, reply, route) => {
			const authorization = req.headers.authorization;
			if (!authorization) {
				reply.status(401).send({ code: "UNAUTHORIZED" });
			}
		},
	],
});
```
