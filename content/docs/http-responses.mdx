## JSON Response

```ts
import { route } from "@rest-rpc/core";
import z from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		create({ title }) {
			return createTodo({ title });
		},
	},
});
```

```ts
const todo = await client.todos.create.fetch({
	title: "Write docs",
});
```

## Response With Multiple Status Codes

Use `fetchResponse()` when the client needs to handle multiple declared statuses.

```ts
export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, todoSchema)
			.response(
				404,
				z.object({
					code: z.literal("TODO_NOT_FOUND"),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		get({ id }) {
			const todo = getTodo(id);

			if (!todo) {
				throw new RouteResponseError(api.todos.get, {
					status: 404,
					body: { code: "TODO_NOT_FOUND" },
				});
			}

			return todo;
		},
	},
});
```

```ts
const response = await client.todos.get.fetchResponse({
	id: "todo_1",
});

if (response.declared && response.status === 200) {
	response.body.title;
}

if (response.declared && response.status === 404) {
	response.body.code;
}
```

## Response Without Body

Call `.response(status)` without a schema to declare a response without a body.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	todos: {
		remove: route
			.delete("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(204),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		remove({ id }) {
			removeTodo(id);
		},
	},
});
```

```ts
await client.todos.remove.fetch({
	id: "todo_1",
});
```

## Response With Typed Headers

Headers declared in the contract are returned from `responseHeaders`.

```ts
export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, {
				body: todoSchema,
				headers: z.object({
					etag: z.string(),
					"x-next-cursor": z.string().optional(),
				}),
			}),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		get({ id }) {
			const todo = getTodo(id);

			return {
				status: 200,
				body: todo,
				responseHeaders: {
					etag: todo.etag,
					"x-next-cursor": todo.nextCursor,
				},
			};
		},
	},
});
```

```ts
const response = await client.todos.get.fetchResponse({
	id: "todo_1",
});

if (response.declared && response.status === 200) {
	response.responseHeaders.etag;
	response.responseHeaders["x-next-cursor"];
}
```

## Response With Undeclared Headers

Undeclared headers are raw HTTP headers returned from `headers`.

```ts
export const api = {
	health: route.get("/health").response(
		200,
		z.object({
			ok: z.literal(true),
		}),
	),
};
```

```ts
export const routes = router(api, {
	health() {
		return {
			status: 200,
			body: { ok: true },
			headers: {
				"cache-control": "no-store",
				"x-request-id": createRequestId(),
			},
		};
	},
});
```

```ts
const response = await client.health.fetchResponse();

if (response.declared) {
	response.headers.get("cache-control");
	response.headers.get("x-request-id");
}
```

## Cookie Response

Writing `Set-Cookie` headers is simplified with `setCookie()` and `clearCookie()`.

```ts
export const api = {
	sessions: {
		create: route
			.post("/sessions")
			.body(
				z.object({
					email: z.string().email(),
					password: z.string(),
				}),
			)
			.response(
				201,
				z.object({
					ok: z.literal(true),
				}),
			),
	},
};
```

```ts
import { clearCookie, setCookie } from "@rest-rpc/express";

export const routes = router(api, {
	sessions: {
		create({ email, password }) {
			const sessionId = createSession({ email, password });

			return {
				status: 201,
				body: { ok: true },
				headers: {
					"set-cookie": [
						clearCookie("old_session", { path: "/" }),
						setCookie("session", sessionId, {
							httpOnly: true,
							secure: true,
							sameSite: "lax",
							path: "/",
						}),
					],
				},
			};
		},
	},
});
```

```ts
await client.sessions.create.fetch({
	email: "ada@example.com",
	password: "password123",
});
```

## Response With Custom Content Type

```ts
import { route } from "@rest-rpc/core";

export const api = {
	reports: {
		csv: route.get("/reports.csv").customResponse(200, {
			contentType: "text/csv",
			schema: z.string(),
		}),
	},
};
```

```ts
export const routes = router(api, {
	reports: {
		csv() {
			return "id,title\n1,First\n";
		},
	},
});
```

```ts
const response = await client.reports.csv.fetch();

await response.text();
```

## Response With Multiple Custom Content Types

When multiple content types are declared, return the selected `contentType` with the response `payload`.

```ts
export const api = {
	images: {
		download: route
			.get("/images/:id")
			.params(z.object({ id: z.string() }))
			.customResponse(200, {
				contentType: ["image/png", "image/jpeg"],
				schema: z.instanceof(Uint8Array),
			}),
	},
};
```

```ts
export const routes = router(api, {
	images: {
		download({ id }) {
			const image = getImage(id);

			return {
				status: 200,
				body: {
					contentType: image.type,
					payload: image.bytes,
				},
			};
		},
	},
});
```

```ts
const response = await client.images.download.fetchResponse({
	id: "image_1",
});

if (response.declared && response.status === 200) {
	if (response.responseHeaders["content-type"] === "image/png") {
		const pngBytes = await response.arrayBuffer();
	} else {
		const jpegBytes = await response.arrayBuffer();
	}
}
```

## Server-Sent Event Responses

Start with `route.sse(path)` for a server-sent event route. The builder declares
one event schema with `.response(schema)`, and the server yields `sseEvent()`
values.

```ts
import { route } from "@rest-rpc/core";
import z from "zod";

const notificationSchema = z.object({
	id: z.string(),
	title: z.string(),
});

export const api = {
	notifications: {
		live: route.sse("/notifications/live").response(notificationSchema),
	},
};
```

```ts
import { router, sseEvent } from "@rest-rpc/express";

export const routes = router(api, {
	notifications: {
		async *live({ context }) {
			const { lastEventId, signal } = context;

			for await (const notification of subscribeNotifications({
				lastEventId,
				signal,
			})) {
				yield sseEvent(notification, {
					id: notification.id,
					retry: 5_000,
				});
			}
		},
	},
});
```

```ts
const source = client.notifications.live.openConnection();

const unsubscribe = source.onMessage((notification) => {
	notification.title;
});

source.onError(() => {
	unsubscribe();
	source.close();
});
```

## Streaming NDJSON Responses

`.streamResponse(status, schema)` sends newline-delimited JSON and the client
reads parsed items.

`request.context.signal` is aborted when the HTTP request is cancelled.

```ts
import { route } from "@rest-rpc/core";

const tokenSchema = z.object({
	index: z.number(),
	text: z.string(),
});

export const api = {
	ai: {
		tokens: route
			.get("/ai/tokens")
			.query(
				z.object({
					prompt: z.string().min(1),
				}),
			)
			.streamResponse(200, tokenSchema),
	},
};
```

```ts
async function* streamTokens(prompt: string, signal: AbortSignal) {
	for await (const token of llm.stream({ prompt, signal })) {
		yield {
			index: token.index,
			text: token.text,
		};
	}
}

export const routes = router(api, {
	ai: {
		tokens(request) {
			const { signal } = request.context;

			return streamTokens(request.prompt, signal);
		},
	},
});
```

```ts
const controller = new AbortController();
const tokens = await client.ai.tokens.fetch(
	{ prompt: "Write a changelog entry" },
	{ signal: controller.signal },
);

let fullText = "";
for await (const token of tokens) {
	fullText += token.text;

	if (token.text.includes("</done>")) {
		controller.abort();
		break;
	}
}
```

## Streaming Responses With Custom Content Type

`.customStreamResponse()` keeps a custom-content response as a raw `Response`
stream.

```ts
export const api = {
	files: {
		download: route
			.get("/files/:id")
			.params(z.object({ id: z.string() }))
			.customStreamResponse(200, {
				contentType: "application/octet-stream",
				schema: z.instanceof(Uint8Array),
			}),
	},
};
```

```ts
export const routes = router(api, {
	files: {
		download(request) {
			const { signal } = request.context;

			return readLargeFile(request.id, signal);
		},
	},
});
```

```ts
const controller = new AbortController();
const response = await client.files.download.fetch(
	{ id: "file_1" },
	{ signal: controller.signal },
);
const reader = response.body?.getReader();
const file = await openWritableFile("archive.zip");
let downloaded = 0;

try {
	while (reader) {
		const { value, done } = await reader.read();
		if (done) break;

		downloaded += value.byteLength;
		await file.write(value);

		if (downloaded > maxDownloadBytes) {
			controller.abort();
			break;
		}
	}
} finally {
	await file.close();
}
```
