## JSON Response

```ts
import { route } from "@rest-rpc/core";
import z from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		create({ body: { title } }) {
			return createTodo({ title });
		},
	},
});
```

```ts
const response = await client.todos.create({
	body: { title: "Write docs" },
});

if (response.status === 201) {
	const todo = response.body;
}
```

## Response With Multiple Status Codes

Call the route and narrow its result by `status` when the client needs to handle
multiple declared statuses.

```ts
export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, todoSchema)
			.response(
				404,
				z.object({
					code: z.literal("TODO_NOT_FOUND"),
				}),
			),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		get({ params: { id } }) {
			const todo = getTodo(id);

			if (!todo) {
				throw new RouteResponseError(api.todos.get, {
					status: 404,
					body: { code: "TODO_NOT_FOUND" },
				});
			}

			return todo;
		},
	},
});
```

```ts
const response = await client.todos.get({
	params: { id: "todo_1" },
});

if (response.status === 200) {
	response.body.title;
}

if (response.status === 404) {
	response.body.code;
}
```

## Response Without Body

Call `.response(status)` without a schema to declare a response without a body.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	todos: {
		remove: route
			.delete("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(204),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		remove({ params: { id } }) {
			removeTodo(id);
		},
	},
});
```

```ts
await client.todos.remove({
	params: { id: "todo_1" },
});
```

## Response With Typed Headers

Headers declared in the contract are returned from `responseHeaders`.

```ts
export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, {
				body: todoSchema,
				headers: z.object({
					etag: z.string(),
					"x-next-cursor": z.string().optional(),
				}),
			}),
	},
};
```

```ts
export const routes = router(api, {
	todos: {
		get({ params: { id } }) {
			const todo = getTodo(id);

			return {
				status: 200,
				body: todo,
				responseHeaders: {
					etag: todo.etag,
					"x-next-cursor": todo.nextCursor,
				},
			};
		},
	},
});
```

```ts
const response = await client.todos.get({
	params: { id: "todo_1" },
});

if (response.status === 200) {
	response.responseHeaders.etag;
	response.responseHeaders["x-next-cursor"];
}
```

## Response With Custom Content Type

```ts
import { route } from "@rest-rpc/core";

export const api = {
	reports: {
		csv: route.get("/reports.csv").customResponse(200, {
			contentType: "text/csv",
			schema: z.string(),
		}),
	},
};
```

```ts
export const routes = router(api, {
	reports: {
		csv() {
			return "id,title\n1,First\n";
		},
	},
});
```

```ts
const response = await client.reports.csv();

if (response.status === 200) {
	await response.body.text();
}
```

## Response With Multiple Custom Content Types

When multiple content types are declared, return the selected `contentType` with the response `payload`.

```ts
export const api = {
	images: {
		download: route
			.get("/images/:id")
			.params(z.object({ id: z.string() }))
			.customResponse(200, {
				contentType: ["image/png", "image/jpeg"],
				schema: z.instanceof(Uint8Array),
			}),
	},
};
```

```ts
export const routes = router(api, {
	images: {
		download({ params: { id } }) {
			const image = getImage(id);

			return {
				status: 200,
				body: {
					contentType: image.type,
					payload: image.bytes,
				},
			};
		},
	},
});
```

```ts
const response = await client.images.download({
	params: { id: "image_1" },
});

if (response.status === 200) {
	if (response.contentType === "image/png") {
		const pngBytes = await response.body.arrayBuffer();
	} else {
		const jpegBytes = await response.body.arrayBuffer();
	}
}
```

## Streaming NDJSON Responses

`.streamResponse(status, schema)` sends newline-delimited JSON and the client
reads parsed items.

`request.context.signal` is aborted when the HTTP request is cancelled.

```ts
import { route } from "@rest-rpc/core";

const tokenSchema = z.object({
	index: z.number(),
	text: z.string(),
});

export const api = {
	ai: {
		tokens: route
			.get("/ai/tokens")
			.query(
				z.object({
					prompt: z.string().min(1),
				}),
			)
			.streamResponse(200, tokenSchema),
	},
};
```

```ts
async function* streamTokens(prompt: string, signal: AbortSignal) {
	for await (const token of llm.stream({ prompt, signal })) {
		yield {
			index: token.index,
			text: token.text,
		};
	}
}

export const routes = router(api, {
	ai: {
		tokens(request) {
			const { signal } = request.context;

			return streamTokens(request.prompt, signal);
		},
	},
});
```

```ts
const controller = new AbortController();
const response = await client.ai.tokens(
	{ prompt: "Write a changelog entry" },
	{ signal: controller.signal },
);
if (response.status !== 200) throw new Error("Unexpected status");
const tokens = response.body;

let fullText = "";
for await (const token of tokens) {
	fullText += token.text;

	if (token.text.includes("</done>")) {
		controller.abort();
		break;
	}
}
```

## Streaming Responses With Custom Content Type

`.customStreamResponse()` keeps a custom-content response as a raw `Response`
stream.

```ts
export const api = {
	files: {
		download: route
			.get("/files/:id")
			.params(z.object({ id: z.string() }))
			.customStreamResponse(200, {
				contentType: "application/octet-stream",
				schema: z.instanceof(Uint8Array),
			}),
	},
};
```

```ts
export const routes = router(api, {
	files: {
		download(request) {
			const { signal } = request.context;

			return readLargeFile(request.params.id, signal);
		},
	},
});
```

```ts
const controller = new AbortController();
const response = await client.files.download(
	{ params: { id: "file_1" } },
	{ signal: controller.signal },
);
if (response.status !== 200) throw new Error("Unexpected status");
const reader = response.body.body?.getReader();
const file = await openWritableFile("archive.zip");
let downloaded = 0;

try {
	while (reader) {
		const { value, done } = await reader.read();
		if (done) break;

		downloaded += value.byteLength;
		await file.write(value);

		if (downloaded > maxDownloadBytes) {
			controller.abort();
			break;
		}
	}
} finally {
	await file.close();
}
```
