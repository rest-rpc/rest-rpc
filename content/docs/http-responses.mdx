## JSON Response

```ts
import { route } from "@rest-rpc/core";
import z from "zod";

const todoSchema = z.object({
	id: z.string(),
	title: z.string(),
});

export const api = {
	todos: {
		create: route
			.post("/todos")
			.body(
				z.object({
					title: z.string().min(1),
				}),
			)
			.response(201, todoSchema),
	},
};
```

```ts
export const routes = {
	todos: {
		create: implement(api.todos.create).handler(({ body: { title } }) => {
			return { status: 201, body: createTodo({ title }) };
		}),
	},
};
```

```ts
const response = await client.todos.create({
	body: { title: "Write docs" },
});

if (response.status === 201) {
	const todo = response.body;
}
```

## Response With Multiple Status Codes

Call the route and narrow its result by `status` when the client needs to handle
multiple declared statuses.

```ts
export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, todoSchema)
			.response(
				404,
				z.object({
					code: z.literal("TODO_NOT_FOUND"),
				}),
			),
	},
};
```

```ts
export const routes = {
	todos: {
		get: implement(api.todos.get).handler(({ params: { id } }) => {
			const todo = getTodo(id);

			if (!todo) {
				throw new RouteResponseError(api.todos.get, {
					status: 404,
					body: { code: "TODO_NOT_FOUND" },
				});
			}

			return { status: 200, body: todo };
		}),
	},
};
```

```ts
const response = await client.todos.get({
	params: { id: "todo_1" },
});

if (response.status === 200) {
	response.body.title;
}

if (response.status === 404) {
	response.body.code;
}
```

## Response Without Body

Call `.response(status)` without a schema to declare a response without a body.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	todos: {
		remove: route
			.delete("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(204),
	},
};
```

```ts
export const routes = {
	todos: {
		remove: implement(api.todos.remove).handler(({ params: { id } }) => {
			removeTodo(id);
			return { status: 204 };
		}),
	},
};
```

```ts
await client.todos.remove({
	params: { id: "todo_1" },
});
```

## Response With Typed Headers

Headers declared in the contract are returned from `responseHeaders`.

```ts
export const api = {
	todos: {
		get: route
			.get("/todos/:id")
			.params(z.object({ id: z.string() }))
			.response(200, {
				body: todoSchema,
				headers: z.object({
					etag: z.string(),
					"x-next-cursor": z.string().optional(),
				}),
			}),
	},
};
```

```ts
export const routes = {
	todos: {
		get: implement(api.todos.get).handler(({ params: { id } }) => {
			const todo = getTodo(id);

			return {
				status: 200,
				body: todo,
				responseHeaders: {
					etag: todo.etag,
					"x-next-cursor": todo.nextCursor,
				},
			};
		}),
	},
};
```

```ts
const response = await client.todos.get({
	params: { id: "todo_1" },
});

if (response.status === 200) {
	response.responseHeaders.etag;
	response.responseHeaders["x-next-cursor"];
}
```

## Response With Custom Content Type

The fetch client parses non-stream custom responses from their declared content
type and returns the schema output. JSON, URL-encoded form data, multipart form
data, text, and binary responses use the corresponding Fetch body reader.

```ts
import { route } from "@rest-rpc/core";

export const api = {
	reports: {
		csv: route.get("/reports.csv").response(200, {
			contentType: "text/csv",
			body: z.string(),
		}),
	},
};
```

```ts
export const routes = {
	reports: {
		csv: implement(api.reports.csv).handler(() => {
			return { status: 200, body: "id,title\n1,First\n" };
		}),
	},
};
```

```ts
const response = await client.reports.csv();

if (response.status === 200) {
	response.body; // string
}
```

## Response With Multiple Custom Content Types

When a response can be one of multiple content types, declare them in an array.
The server handler must specify the `contentType` of the response.

```ts
export const api = {
	images: {
		download: route
			.get("/images/:id")
			.params(z.object({ id: z.string() }))
			.response(200, {
				contentType: ["image/png", "image/jpeg"],
				body: z.instanceof(Uint8Array),
			}),
	},
};
```

```ts
export const routes = {
	images: {
		download: implement(api.images.download).handler(({ params: { id } }) => {
			const image = getImage(id);

			return {
				status: 200,
				body: image.bytes,
				contentType: image.type,
			};
		}),
	},
};
```

```ts
const response = await client.images.download({
	params: { id: "image_1" },
});

if (response.status === 200) {
	if (response.contentType === "image/png") {
		const pngBytes = response.body;
	} else {
		const jpegBytes = response.body;
	}
}
```

## Streaming NDJSON Responses

`.streamResponse(status, schema)` sends newline-delimited JSON and the client
reads parsed items.

The handler's `signal` is aborted when the HTTP request is cancelled.

```ts
import { route } from "@rest-rpc/core";

const tokenSchema = z.object({
	index: z.number(),
	text: z.string(),
});

export const api = {
	ai: {
		tokens: route
			.get("/ai/tokens")
			.query(
				z.object({
					prompt: z.string().min(1),
				}),
			)
			.streamResponse(200, tokenSchema),
	},
};
```

```ts
async function* streamTokens(prompt: string, signal: AbortSignal) {
	for await (const token of llm.stream({ prompt, signal })) {
		yield {
			index: token.index,
			text: token.text,
		};
	}
}

export const routes = {
	ai: {
		tokens: implement(api.ai.tokens).handler(({ query, signal }) => ({
			status: 200,
			body: streamTokens(query.prompt, signal),
		})),
	},
};
```

```ts
const controller = new AbortController();
const response = await client.ai.tokens(
	{ prompt: "Write a changelog entry" },
	{ signal: controller.signal },
);
if (response.status !== 200) throw new Error("Unexpected status");
const tokens = response.body;

let fullText = "";
for await (const token of tokens) {
	fullText += token.text;

	if (token.text.includes("</done>")) {
		controller.abort();
		break;
	}
}
```
